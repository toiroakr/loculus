export class ValidationError extends Error {
  constructor(readonly issues: readonly Issue[]) {
    super(
      issues
        .map(
          (i) =>
            `${i.path?.map((p) => (typeof p === "object" ? String(p.key) : String(p))).join(".") ?? ""}: ${i.message}`,
        )
        .join("; "),
    );
    this.name = "ValidationError";
  }
}
export function parse<S extends AnySchema>(schema: S, value: unknown): Output<S> {
  const result = safeParse(schema, value);
  if (result.issues) throw new ValidationError(result.issues);
  return result.value;
}
export type DefaultTargets<T> =
  | { readonly nullish: T; readonly undefined?: never; readonly null?: never }
  | { readonly undefined: T; readonly null?: T; readonly nullish?: never }
  | { readonly null: T; readonly undefined?: T; readonly nullish?: never };
export interface FieldOptions<T> {
  readonly description?: string;
  readonly optional?: boolean;
  readonly nullable?: boolean;
  readonly nullish?: boolean;
  readonly default?: T | DefaultTargets<T>;
}
type BasicPresence = { description?: string; nullable?: false; nullish?: false; default?: never };
type FlagValue<P, K extends PropertyKey> = K extends keyof P ? P[K] : never;
type DefaultFlags<P> = P extends { default: infer D }
  ? D extends undefined ? [false, false]
    : D extends readonly unknown[] | Date ? [true, false]
    : D extends { nullish: unknown } ? [true, true]
    : D extends { undefined: unknown } ? [true, D extends { null: unknown } ? true : false]
    : D extends { null: unknown } ? [false, true]
    : [true, false]
  : "default" extends keyof P ? [boolean, boolean] : [false, false];
type PresenceFlags<P> = [
  true extends FlagValue<P, "optional"> | FlagValue<P, "nullish"> | DefaultFlags<P>[0] ? true : false,
  true extends FlagValue<P, "nullable"> | FlagValue<P, "nullish"> | DefaultFlags<P>[1] ? true : false,
];
type ConfiguredInput<I, P> = I
  | (PresenceFlags<P>[0] extends true ? undefined : never)
  | (PresenceFlags<P>[1] extends true ? null : never);
type UndefinedDefault<P> = P extends { default: infer D }
  ? D extends readonly unknown[] | Date ? never
    : D extends { nullish: infer V } ? undefined extends V ? undefined : never
    : D extends object ? undefined extends D[keyof D & ("undefined" | "null")] ? undefined : never
    : never
  : never;
type ConfiguredOutput<O, P> = O | UndefinedDefault<P>
  | (PresenceFlags<P>[0] extends true ? false extends DefaultFlags<P>[0] ? undefined : never : never)
  | (PresenceFlags<P>[1] extends true ? false extends DefaultFlags<P>[1] ? null : never : never);
type ConfiguredField<I, O, P> = Field<ConfiguredInput<I, P>, ConfiguredOutput<O, P>>;
type FieldPlan = FieldOptions<any> & StringOptions & NumberOptions & ArrayOptions & { kind: Meta["kind"]; inner?: (value: unknown) => Result<any> };

function validateString(value: unknown): Result<string> {
  return typeof value === "string" ? { value } : fail("Expected a string", "invalid_type");

type PlannedValidate = ((value: unknown) => Result<any>) & { [fieldPlan]?: FieldPlan };
function configuredField<I, O>(
  meta: Meta,
  options: FieldOptions<O>,
  validate: (value: unknown) => Result<O>,
): Field<any, any, boolean> {
  if (options === emptyOptions) return { meta, "~standard": { version: 1, vendor: "loculus", validate }, field: true } as Field<any, any, boolean>;
  const value = own(options, "default") ? options.default : undefined;
  if (options.optional || options.nullable || options.nullish || value !== undefined) {
    let defaults = 0;
    let missing: O | undefined;
    let nil: O | undefined;
    if (value !== undefined) {
      if (typeof value === "object" && value !== null && !Array.isArray(value) && !(value instanceof Date)) {
        const keys = Object.keys(value);
        const nullish = own(value, "nullish");
        if (!keys.length || keys.some(key => key !== "undefined" && key !== "null" && key !== "nullish") || (nullish && keys.length > 1))
          throw new TypeError("Invalid default targets");
        defaults = (nullish || own(value, "undefined") ? 1 : 0) | (nullish || own(value, "null") ? 2 : 0);
        missing = nullish ? (value as DefaultTargets<O>).nullish : (value as DefaultTargets<O>).undefined;
        nil = nullish ? (value as DefaultTargets<O>).nullish : (value as DefaultTargets<O>).null;
      } else {
        defaults = 1;
        missing = value as O;
      }
    }
    const optional = !!(options.optional || options.nullish || (defaults & 1));
    const nullable = !!(options.nullable || options.nullish || (defaults & 2));
    const base = validate;
    if (optional || nullable) validate = (input: unknown): Result<any> => {
      if (input === undefined && optional) return { value: missing };
      if (input === null && nullable) return { value: defaults & 2 ? nil : null };
      return base(input);
    };
    (meta as { optional: boolean }).optional = optional;
    (meta as { nullable: boolean }).nullable = nullable;
    if (defaults & 1) (meta as { defaultValue?: O }).defaultValue = missing;
  }
  return { meta, "~standard": { version: 1, vendor: "loculus", validate }, field: true } as Field<any, any, boolean>;
}

/* @__NO_SIDE_EFFECTS__ */
export function string(
  options: StringOptions & FieldOptions<string> = emptyOptions,
): Field<any, any, boolean> {
  if (options !== emptyOptions) options = { ...options };
  const result = configuredField<string, string>(
    { kind: "string", description: options.description },
    options,
    !options.trim && options.min === undefined && options.max === undefined && !options.pattern
      ? validateString
      : (value) => {
      if (typeof value !== "string") return fail("Expected a string", "invalid_type");
      const output = options.trim ? value.trim() : value;
      if (options.min !== undefined && output.length < options.min)
        return fail(`Expected at least ${options.min} characters`);
      if (options.max !== undefined && output.length > options.max)
        return fail(`Expected at most ${options.max} characters`);
      if (options.pattern) {
        const pattern = options.pattern;
        const previous = pattern.lastIndex;
        pattern.lastIndex = 0;
        const valid = pattern.test(output);
        pattern.lastIndex = previous;
        if (!valid) return fail("Invalid format");
      }
      return { value: output };
    },
  );
  const validate = result["~standard"].validate as PlannedValidate;
  if (validate !== validateString) {
    const plan = options as FieldPlan;
    plan.kind = "string";
    validate[fieldPlan] = plan;
  }
  return result;
}

  options: NumberOptions & FieldOptions<number> = emptyOptions,

  if (options !== emptyOptions) options = { ...options };
  const result = configuredField<string | number, number>(
    { kind: "number", description: options.description },

    (value) => {
      if (typeof value !== "number") {

          return fail("Expected a decimal number", "invalid_type");

        return fail(`Expected greater than ${options.gt}`);
      if (options.gte !== undefined && output < options.gte)
        return fail(`Expected at least ${options.gte}`);
      if (options.lt !== undefined && output >= options.lt)
        return fail(`Expected less than ${options.lt}`);
      if (options.lte !== undefined && output > options.lte)
        return fail(`Expected at most ${options.lte}`);
      return { value: output };
    },
  );
  const validate = result["~standard"].validate as PlannedValidate;
  const plan = (options === emptyOptions ? {} : options) as FieldPlan;
  plan.kind = "number";
  validate[fieldPlan] = plan;
  return result;
}

export function enum_<const T extends readonly [string, ...string[]]>(
  values: T,
  options: FieldOptions<T[number]> = emptyOptions,
): Field<any, any, boolean> {
  if (options !== emptyOptions) options = { ...options };
  const allowed = [...values];
  let message: string;
  let lookup: Set<string> | undefined;
  let uses = 0;
  const contains = allowed.length === 2
    ? (value: unknown) => value === allowed[0] || value === allowed[1]
    : (value: unknown) => {
      if (!lookup && allowed.length > 8 && ++uses === 8) lookup = new Set(allowed);
      return lookup ? lookup.has(value as string) : allowed.includes(value as T[number]);
    };
  return configuredField<T[number], T[number]>(
    new EnumMeta(allowed, options.description),
    options,
    (value) => {
      return typeof value === "string" && contains(value)
        ? { value: value as T[number] }
        : fail(message ??= `Expected one of: ${allowed.join(", ")}`);
    },
  );
}

function validateDate(value: unknown): Result<Date> {
  if (typeof value !== "string" || !/^\d{4}-\d{2}-\d{2}(?:T.*)?$/.test(value))
    return fail("Expected an ISO date");
  const output = new Date(value);
  if (!Number.isFinite(output.getTime())) return fail("Invalid date");
  const year = +value.slice(0, 4);
  const month = +value.slice(5, 7);
  const day = +value.slice(8, 10);
  const days = month === 2 ? (year % 4 === 0 && (year % 100 !== 0 || year % 400 === 0) ? 29 : 28) : 31 - ((((1 << 4) | (1 << 6) | (1 << 9) | (1 << 11)) >> month) & 1);
  if (month < 1 || month > 12 || day < 1 || day > days) return fail("Invalid date");
  return { value: output };
}

export function array<I, O>(
  s: ArrayElement<I, O>,
  options: ArrayOptions & FieldOptions<O[]> = emptyOptions,
): Field<any, any, boolean> {
  if (options !== emptyOptions) options = { ...options };
  if (s.meta.kind === "array") throw new TypeError("Nested arrays are unsupported");
  const innerValidate = s["~standard"].validate;
  const result = configuredField<I | readonly I[], O[]>(
    { kind: "array", inner: s, enumValues: s.meta.enumValues, description: options.description },
    options,
    (value) => {
      if (options.wrapScalar === false && !Array.isArray(value))
        return fail("Expected an array", "invalid_type");
      const inputs = Array.isArray(value) ? value : [value];
      if (options.min !== undefined && inputs.length < options.min)
        return fail(`Expected at least ${options.min} items`);
      if (options.max !== undefined && inputs.length > options.max)
        return fail(`Expected at most ${options.max} items`);
      const output: O[] = [];
      let issues: Issue[] | undefined;
      for (let i = 0; i < inputs.length; i++) {
        const result = innerValidate(inputs[i]);
        if (result.issues)
          for (const issue of result.issues)
            (issues ??= []).push({ ...issue, path: issue.path ? [i, ...issue.path] : [i] });
        else output.push(result.value);
      }
      return issues ? { issues } : { value: output };
    },
  );
  const validate = result["~standard"].validate as PlannedValidate;
  const plan = (options === emptyOptions ? {} : options) as FieldPlan;
  plan.kind = "array";
  plan.inner = innerValidate;
  validate[fieldPlan] = plan;
  return result;
}

  s: S,
  check: (value: Output<S>) => boolean,

const own = Object.hasOwn;
function validateObject(input: unknown, fields: Shape, keys: readonly string[], validators: readonly ((value: unknown) => Result<any>)[], mode: "strip" | "strict" | "passthrough"): Result<Record<string, unknown>> {
    if (typeof input !== "object" || input === null || Array.isArray(input))
      return fail("Expected an object", "invalid_type");
    const raw = input as Record<string, unknown>;
    const output: Record<string, unknown> = {};
    let issues: Issue[] | undefined;
    for (let i = 0; i < keys.length; i++) {
      const key = keys[i]!;
      const result = validators[i]!(own(raw, key) ? raw[key] : undefined);
      if (result.issues) {
        issues ??= [];
        for (const issue of result.issues)
          issues.push({ ...issue, path: issue.path ? [key, ...issue.path] : [key] });
      } else if (result.value !== undefined || own(raw, key)) {
        if (key === "__proto__")
          Object.defineProperty(output, key, { value: result.value, enumerable: true, writable: true, configurable: true });
        else output[key] = result.value;
      }
    }
    if (mode === "strict") {
      for (const key in raw)
        if (!own(fields, key) && own(raw, key))
          (issues ??= []).push({ message: `Unknown key: ${key}`, code: "unrecognized_key", path: [key] });
    } else if (mode === "passthrough") {
      for (const key of Object.keys(raw)) {
        if (!own(fields, key)) {
          const value = raw[key];
          if (key in output)
            Object.defineProperty(output, key, {
              value, enumerable: true, writable: true, configurable: true,
            });
          else output[key] = value;
        }
      }
    }
    return issues ? { issues } : { value: output };
}
/* @__NO_SIDE_EFFECTS__ */
export function object<S extends Shape>(
  shape: S,
  options: { unknownKeys?: "strip" | "strict" | "passthrough"; description?: string } = emptyOptions,
): ObjectSchema<S> {
  const fields = Object.freeze({ ...shape });
  const keys = Object.keys(fields);
  const validators: ((value: unknown) => Result<any>)[] = [];
  for (const key of keys) {
    const s = fields[key]!;
    if (s.field !== true) throw new TypeError(`Expected field: ${key}`);
    validators.push(s["~standard"].validate);
  }
  const mode = options.unknownKeys ?? "strip";
  const validate = (input: unknown): Result<ObjectOutput<S>> => validateObject(input, fields, keys, validators, mode) as Result<ObjectOutput<S>>;
  return {
    meta: { kind: "object", shape: fields, unknownKeys: mode, description: options.description },
    "~standard": { version: 1, vendor: "loculus", validate },
    shape: fields as S,
  } as unknown as ObjectSchema<S>;
}
function appendCompiledIssues(issues: Issue[] | undefined, errors: readonly Issue[], key: PropertyKey): Issue[] {
  issues ??= [];
  for (const issue of errors) issues.push({ ...issue, path: issue.path ? [key, ...issue.path] : [key] });
  return issues;
}
function compiledCheck(plan: FieldPlan | undefined, value: string): string {
  if (!plan) return "false";
  const options = plan;
  const checks: string[] = [];
  if ((["min", "max", "gt", "gte", "lt", "lte"] as const).some((key) => {
    const bound = options[key];
    return bound !== undefined && typeof bound !== "number";
  })) return "false";
  if (plan.kind === "string") {
    if (options.trim || options.pattern) return "false";
    checks.push(`typeof ${value}==="string"`);
    if (options.min !== undefined) checks.push(`!(${value}.length<${String(options.min)})`);
    if (options.max !== undefined) checks.push(`!(${value}.length>${String(options.max)})`);
  } else if (plan.kind === "number") {
    const finite = options.integer ? `Number.isSafeInteger(${value})` : `Number.isFinite(${value})`;
    checks.push(`(typeof ${value}==="string"?(${value}.length===1?(${value}>="0"&&${value}<="9"&&(${value}=+${value},true)):decimal.test(${value})&&(${value}=+${value},${finite})):typeof ${value}==="number"&&${finite})`);
    for (const [key, operator] of [["gt", "<="], ["gte", "<"], ["lt", ">="], ["lte", ">"]] as const)
      if (options[key] !== undefined) {
        const bound = options[key];
        if (options.integer && ((key === "gt" && bound < -Number.MAX_SAFE_INTEGER) ||
            (key === "gte" && bound <= -Number.MAX_SAFE_INTEGER) ||
            (key === "lt" && bound > Number.MAX_SAFE_INTEGER) ||
            (key === "lte" && bound >= Number.MAX_SAFE_INTEGER))) continue;
        checks.push(`!(${value}${operator}${String(bound)})`);
      }
  } else if (plan.kind === "boolean") {
    return `typeof ${value}==="boolean"||(${value}==="false"||${value}==="0"?(${value}=false,true):${value}==="true"||${value}==="1"?(${value}=true,true):false)`;
  } else return "false";
  return checks.join("&&");
}
function compiledRemainder(
  raw: Record<string, unknown>,
  start: number,
  output: Record<string, unknown>,
  keys: readonly string[],
  validators: readonly ((value: unknown) => Result<any>)[],
  first?: Result<any>,
  firstRead?: { readonly issues?: readonly Issue[]; readonly value: unknown },
): Result<Record<string, unknown>> {
  let issues: Issue[] | undefined;
  for (let i = start; i < keys.length; i++) {
    const key = keys[i]!;
    const result = i === start && first ? first : validators[i]!(
      i === start ? output[key] : own(raw, key) ? raw[key] : undefined,
    );
    if (i === start && firstRead ? firstRead.issues : result.issues) issues = appendCompiledIssues(issues, result.issues!, key);
    else if ((i === start && firstRead ? firstRead.value : (result as { value: any }).value) !== undefined || own(raw, key)) output[key] = (result as { value: any }).value;
    else delete output[key];
  }
  return issues ? { issues } : { value: output };
}
function compileArray(plan: FieldPlan, validate: (value: unknown) => Result<any>): (value: unknown) => Result<any> {
  const options = plan;
  const inner = plan.inner!;
  const innerPlan = inner === validateString ? { kind: "string" as const } : inner === validateBoolean ? { kind: "boolean" as const } : (inner as PlannedValidate)[fieldPlan];
  return new Function("base", "inner", "options", "fail", "decimal", "append", `return function(input){
if(input===undefined||input===null)return base(input);
if(${options.wrapScalar === false}&&!Array.isArray(input))return fail("Expected an array","invalid_type");
const inputs=Array.isArray(input)?input:[input];
if(${options.min !== undefined}&&inputs.length<options.min)return fail("Expected at least "+options.min+" items");
if(${options.max !== undefined}&&inputs.length>options.max)return fail("Expected at most "+options.max+" items");
const output=[];let issues;
for(let i=0;i<inputs.length;i++){
const rawItem=inputs[i];let item=rawItem;
if(${compiledCheck(innerPlan, "item")})output.push(item);
else{const result=inner(rawItem);if(result.issues)issues=append(issues,result.issues,i);else output.push(result.value);}
}
return issues?{issues}:{value:output};}`)(validate, inner, options, fail, decimal, appendCompiledIssues);
}
/* @__NO_SIDE_EFFECTS__ */
export function compile<S extends Shape>(s: ObjectSchema<S>): ObjectSchema<S> {
  if (s.meta.unknownKeys !== "strip" || s.meta.inner) return s;
  const keys = Object.keys(s.shape);
  const validators = keys.map((key) => s.shape[key]!["~standard"].validate);
  try {
    const plans = validators.map((validate) => validate === validateString ? { kind: "string" as const } : validate === validateBoolean ? { kind: "boolean" as const } : (validate as PlannedValidate)[fieldPlan]);
    const compiledValidators = validators.map((validate, i) => plans[i]?.kind === "array" ? compileArray(plans[i]!, validate) : validate);
    const wide = keys.length >= 32;
    const checks = wide ? plans.map((plan, i) => plan?.kind === "array" ? "array" : compiledCheck(plan, `v${i}`)) : undefined;
    const suffix = checks ? new Array<number>(keys.length + 1) : undefined;
    if (suffix && checks) {
      suffix[keys.length] = 0;
      for (let i = keys.length - 1; i >= 0; i--) suffix[i] = suffix[i + 1]! + (checks[i] !== "false" ? 1 : 0);
    }
    let continuing = false;
    const statements = keys.map((key, i) => {
      const property = JSON.stringify(key);
      const prefix = `var v${i}=own(raw,${property})?raw[${property}]:undefined;`;
      if (checks ? checks[i] === "array" : plans[i]?.kind === "array") return `${prefix}const r${i}=validators[${i}](v${i});
if(r${i}.issues){first=r${i};start=${i};break fast;}
v${i}=r${i}.value;
if(v${i}===undefined&&!own(raw,${property})){first=r${i};start=${i};break fast;}`;
      const check = checks ? checks[i]! : compiledCheck(plans[i], `v${i}`);
      const plan = plans[i];
      if (checks && suffix![i + 1]! >= 4 && (check === "false" || plan && (plan.optional || plan.nullable || plan.nullish || own(plan, "default")))) {
        continuing = true;
        const fallback = `const r${i}=validators[${i}](v${i});
const e${i}=r${i}.issues;
if(e${i}){first=r${i};firstRead={issues:e${i},value:undefined};start=${i};break fast;}
const o${i}=r${i}.value;
if(o${i}===undefined){first=r${i};firstRead={issues:e${i},value:o${i}};start=${i};break fast;}
v${i}=r${i}.value;`;
        return check === "false" ? `${prefix}${fallback}` : `${prefix}if(!(${check})){${fallback}}`;
      }
      return `${prefix}if(!(${check})){start=${i};break fast;}`;
    });
    const properties = keys.map((key, i) => `${key === "__proto__" ? `[${JSON.stringify(key)}]` : JSON.stringify(key)}:v${i}`).join(",");
    const validate = new Function("fail", "decimal", "remainder", "keys", "validators", "own",
      `return function(raw){
if(typeof raw!=="object"||raw===null||Array.isArray(raw))return fail("Expected an object","invalid_type");
${continuing ? "let start,first,firstRead;" : "let start,first;"}
fast:{${statements.join("\n")}
return {value:{${properties}}};}
return remainder(raw,start,{${properties}},keys,validators,first${continuing ? ",firstRead" : ""});}`,
    )(fail, decimal, compiledRemainder, keys, compiledValidators, own);
    return { ...s, "~standard": { ...s["~standard"], validate } };
  } catch {
    return s;
  }
}

export function union<const S extends readonly [AnySchema, AnySchema, ...AnySchema[]]>(
  options: S,
  config: { description?: string } = {},
): Schema<Input<S[number]>, Output<S[number]>> {
  const branches = Object.freeze([...options]);
  return schema({ kind: "union", options: branches, description: config.description }, (input) => {
    let issues: Issue[] | undefined;
    for (let i = 0; i < branches.length; i++) {
      const result = branches[i]!["~standard"].validate(input);
      if (!result.issues) return result;
      (issues ??= []).push(...result.issues);
    }
    return { issues: issues! };
  });
}
/* @__NO_SIDE_EFFECTS__ */
export function xor<const S extends readonly [AnySchema, AnySchema, ...AnySchema[]]>(
  options: S,
  config: { description?: string } = {},
): Schema<Input<S[number]>, Output<S[number]>> {
  const branches = Object.freeze([...options]);
  return schema({ kind: "xor", options: branches, description: config.description }, (input) => {
    let success: Result<Output<S[number]>> | undefined;
    for (const s of branches) {
      const result = s["~standard"].validate(input);
      if (!result.issues) {
        if (success) return fail("Expected exactly one matching schema");
        success = result;
      }
    }
    return success ?? fail("No matching schema");
  });
}
/* @__NO_SIDE_EFFECTS__ */
export function discriminatedUnion<
  const S extends readonly [ObjectSchema, ObjectSchema, ...ObjectSchema[]],
>(
  key: string,
  options: S,
  config: { description?: string } = {},
): Schema<Input<S[number]>, Output<S[number]>> {
  const lookup = new Map<string, AnySchema>();
  for (const option of options) {
    const values = option.shape[key]?.meta.enumValues;
    if (!values?.length) throw new TypeError("Discriminator must be an enum or literal");
    for (const value of values) {
      if (lookup.has(value)) throw new TypeError(`Duplicate discriminator: ${value}`);
      lookup.set(value, option);
    }
  }
  return schema(
    {
      kind: "union",
      discriminator: key,
      options: Object.freeze([...options]),
      description: config.description,
    },
    (input) => {
      if (typeof input !== "object" || input === null || !own(input, key))
        return fail(`Missing discriminator: ${key}`);
      const selected = lookup.get((input as Record<string, string>)[key]!);
      return selected
        ? selected["~standard"].validate(input)
        : { issues: [{ message: `Invalid discriminator: ${key}`, path: [key] }] };
    },
  );
}
/* @__NO_SIDE_EFFECTS__ */
export function intersection<A extends AnySchema, B extends AnySchema>(
  a: A,
  b: B,
  config: { description?: string } = {},
): Schema<Input<A> & Input<B>, Output<A> & Output<B>> {
  return schema(
    { kind: "intersection", options: [a, b], description: config.description },
    (input) => {
      const left = a["~standard"].validate(input);
      const right = b["~standard"].validate(input);
      if (left.issues || right.issues)
        return { issues: [...(left.issues ?? []), ...(right.issues ?? [])] };
      if (
        typeof left.value !== "object" ||
        left.value === null ||
        typeof right.value !== "object" ||
        right.value === null
      )
        return fail("Intersections require object outputs");
      for (const key of Object.keys(left.value))
        if (own(right.value, key) && !Object.is(left.value[key], right.value[key]))
          return fail(`Conflicting intersection output: ${key}`);
      return { value: { ...left.value, ...right.value } };
    },
  );
}
export function fromEntries(
  entries: Iterable<readonly [string, unknown]>,
): Record<string, unknown> {
  const result: Record<string, unknown> = Object.create(null);
  for (const [key, value] of entries) {
    if (own(result, key)) {
      const current = result[key];
      if (Array.isArray(current)) current.push(value);
      else result[key] = [current, value];
    } else result[key] = value;
  }
  return result;
}
