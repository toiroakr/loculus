export function safeParse<S extends AnySchema>(schema: S, value: unknown): Result<Output<S>> {
  return schema["~standard"].validate(value);
}
export class ValidationError extends Error {
  constructor(readonly issues: readonly Issue[]) {
    super(
      issues
        .map(
          (i) =>
            `${i.path?.map((p) => (typeof p === "object" ? String(p.key) : String(p))).join(".") ?? ""}: ${i.message}`,
        )
        .join("; "),
    );
    this.name = "ValidationError";
  }
}
export function parse<S extends AnySchema>(schema: S, value: unknown): Output<S> {
  const result = safeParse(schema, value);
  if (result.issues) throw new ValidationError(result.issues);
  return result.value;
}
export type DefaultTargets<T> =
  | { readonly nullish: T; readonly undefined?: never; readonly null?: never }
  | { readonly undefined: T; readonly null?: T; readonly nullish?: never }
  | { readonly null: T; readonly undefined?: T; readonly nullish?: never };
export interface FieldOptions<T> {
  readonly description?: string;
  readonly optional?: boolean;
  readonly nullable?: boolean;
  readonly nullish?: boolean;
  readonly default?: T | DefaultTargets<T>;
}
type BasicPresence = { description?: string; nullable?: false; nullish?: false; default?: never };
type Flag<P, K extends PropertyKey> = K extends keyof P
  ? true extends P[K]
    ? true
    : false
  : false;
type HasDefault<P, K extends "undefined" | "null"> = P extends { default: infer D }
  ? D extends undefined
    ? false
    : D extends { nullish: unknown }
      ? true
      : D extends { [Key in K]: unknown }
        ? true
        : D extends DefaultTargets<unknown>
          ? false
          : K extends "undefined"
            ? true
            : false
  : "default" extends keyof P
    ? boolean
    : false;
type AllowsUndefined<P> = true extends
  | Flag<P, "optional">
  | Flag<P, "nullish">
  | HasDefault<P, "undefined">
  ? true
  : false;
type AllowsNull<P> = true extends Flag<P, "nullable"> | Flag<P, "nullish"> | HasDefault<P, "null">
  ? true
  : false;
type ConfiguredInput<I, P> =
  | I
  | (AllowsUndefined<P> extends true ? undefined : never)
  | (AllowsNull<P> extends true ? null : never);
type ConfiguredOutput<O, P> =
  | O
  | (AllowsUndefined<P> extends true
      ? false extends HasDefault<P, "undefined">
        ? undefined
        : never
      : never)
  | (AllowsNull<P> extends true ? (false extends HasDefault<P, "null"> ? null : never) : never);
type ConfiguredField<I, O, P> = Field<ConfiguredInput<I, P>, ConfiguredOutput<O, P>>;
type FieldPlan = { kind: Meta["kind"]; enum: boolean; options: object; inner?: (value: unknown) => Result<any> };
const fieldPlan = Symbol();
type PlannedValidate = ((value: unknown) => Result<any>) & { [fieldPlan]?: FieldPlan };
function plannedField<I, O>(
  meta: Meta,
  options: FieldOptions<O>,
  validate: (value: unknown) => Result<O>,
  innerValidate?: (value: unknown) => Result<any>,
): Field<any, any, boolean> {
  (validate as PlannedValidate)[fieldPlan] = { kind: meta.kind, enum: !!meta.enumValues, inner: innerValidate, options };
  return field(meta, validate);
}
function configuredField<I, O>(
  meta: Meta,
  options: FieldOptions<O>,
  validate: (value: unknown) => Result<O>,
  innerValidate?: (value: unknown) => Result<any>,
): Field<any, any, boolean> {
  if (!(options.optional || options.nullable || options.nullish || own(options, "default")))
    return plannedField(meta, options, validate, innerValidate);
  let hasUndefinedDefault = false;
  let hasNullDefault = false;
  let undefinedDefault: O | undefined;
  let nullDefault: O | undefined;
  if (own(options, "default") && options.default !== undefined) {
    const value = options.default;
    if (
      typeof value === "object" &&
      value !== null &&
      !Array.isArray(value) &&
      !(value instanceof Date)
    ) {
      const targets = value as DefaultTargets<O>;
      const keys = Object.keys(targets);
      if (!keys.length || keys.some((key) => key !== "undefined" && key !== "null" && key !== "nullish"))
        throw new TypeError("Default targets must specify undefined, null or nullish");
      const nullish = own(targets, "nullish");
      if (nullish && keys.length > 1)
        throw new TypeError("nullish defaults cannot be combined with undefined or null defaults");
      hasUndefinedDefault = nullish || own(targets, "undefined");
      hasNullDefault = nullish || own(targets, "null");
      undefinedDefault = nullish ? targets.nullish : targets.undefined;
      nullDefault = nullish ? targets.nullish : targets.null;
    } else {
      hasUndefinedDefault = true;
      undefinedDefault = value as O;
    }
  }
  const optional = options.optional || options.nullish || hasUndefinedDefault;
  const nullable = options.nullable || options.nullish || hasNullDefault;
  const run = !optional && !nullable ? validate : (value: unknown): Result<O | null | undefined> => {
    if (value === undefined && optional)
      return { value: hasUndefinedDefault ? undefinedDefault : undefined };
    if (value === null && nullable) return { value: hasNullDefault ? nullDefault : null };
    return validate(value);
  };
  const resolved = meta as { -readonly [K in keyof Meta]: Meta[K] };
  resolved.optional = optional;
  resolved.nullable = nullable;
  if (hasUndefinedDefault) resolved.defaultValue = undefinedDefault;
  return plannedField(meta, options, run, innerValidate);
}

/* @__NO_SIDE_EFFECTS__ */
export function string(
  options: StringOptions & FieldOptions<string> = {},
): Field<any, any, boolean> {
  options = { ...options };
  return configuredField<string, string>(
    { kind: "string", optional: false, nullable: false, description: options.description },
    options,
    (value) => {
      if (typeof value !== "string") return fail("Expected a string", "invalid_type");
      const output = options.trim ? value.trim() : value;
      if (options.min !== undefined && output.length < options.min)
        return fail(`Expected at least ${options.min} characters`);
      if (options.max !== undefined && output.length > options.max)
        return fail(`Expected at most ${options.max} characters`);
      if (options.pattern) {
        const pattern = options.pattern;
        const previous = pattern.lastIndex;
        pattern.lastIndex = 0;
        const valid = pattern.test(output);
        pattern.lastIndex = previous;
        if (!valid) return fail("Invalid format");
      }
      return { value: output };
    },
  );
}

export function number(
  options: NumberOptions & FieldOptions<number> = {},
): Field<any, any, boolean> {
  options = { ...options };
  return configuredField<string | number, number>(
    { kind: "number", optional: false, nullable: false, description: options.description },
    options,
    (value) => {
      if (typeof value !== "number") {
        if (typeof value !== "string" || !decimal.test(value))
          return fail("Expected a decimal number", "invalid_type");
        value = Number(value);
      }
      const output = value as number;
      if (!Number.isFinite(output)) return fail("Expected a finite number");
      if (options.integer && !Number.isSafeInteger(output)) return fail("Expected a safe integer");
      if (options.gt !== undefined && output <= options.gt)
        return fail(`Expected greater than ${options.gt}`);
      if (options.gte !== undefined && output < options.gte)
        return fail(`Expected at least ${options.gte}`);
      if (options.lt !== undefined && output >= options.lt)
        return fail(`Expected less than ${options.lt}`);
      if (options.lte !== undefined && output > options.lte)
        return fail(`Expected at most ${options.lte}`);
      return { value: output };
    },
  );
}

/* @__NO_SIDE_EFFECTS__ */
export function array<I, O>(
  s: Field<I, O>,
  options: ArrayOptions & FieldOptions<O[]> = {},
): Field<any, any, boolean> {
  options = { ...options };
  if (s.meta.kind === "array") throw new TypeError("Nested arrays are unsupported");
  const innerValidate = s["~standard"].validate;
  return configuredField<I | readonly I[], O[]>(
    { kind: "array", inner: s, enumValues: s.meta.enumValues, optional: false, nullable: false, description: options.description },
    options,
    (value) => {
      if (options.wrapScalar === false && !Array.isArray(value))
        return fail("Expected an array", "invalid_type");
      const inputs = Array.isArray(value) ? value : [value];
      if (options.min !== undefined && inputs.length < options.min)
        return fail(`Expected at least ${options.min} items`);
      if (options.max !== undefined && inputs.length > options.max)
        return fail(`Expected at most ${options.max} items`);
      const output: O[] = [];
      let issues: Issue[] | undefined;
      for (let i = 0; i < inputs.length; i++) {
        const result = innerValidate(inputs[i]);
        if (result.issues)
          for (const issue of result.issues)
            (issues ??= []).push({ ...issue, path: [i, ...(issue.path ?? [])] });
        else output.push(result.value);
      }
      return issues ? { issues } : { value: output };
    },
    innerValidate,
  );
}

const own = Object.hasOwn;
/* @__NO_SIDE_EFFECTS__ */
export function object<S extends Shape>(
  shape: S,
  options: { unknownKeys?: "strip" | "strict" | "passthrough"; description?: string } = {},
): ObjectSchema<S> {
  const fields = Object.freeze({ ...shape });
  const keys = Object.keys(fields);
  const validators: ((value: unknown) => Result<any>)[] = [];
  for (const key of keys) {
    const s = fields[key]!;
    if (s.field !== true) throw new TypeError(`Expected field: ${key}`);
    validators.push(s["~standard"].validate);
  }
  const mode = options.unknownKeys ?? "strip";
  const validate = (input: unknown): Result<ObjectOutput<S>> => {
    if (typeof input !== "object" || input === null || Array.isArray(input))
      return fail("Expected an object", "invalid_type");
    const raw = input as Record<string, unknown>;
    const output: Record<string, unknown> = {};
    let issues: Issue[] | undefined;
    for (let i = 0; i < keys.length; i++) {
      const key = keys[i]!;
      const value = own(raw, key) ? raw[key] : undefined;
      const result = validators[i]!(value);
      if (result.issues) {
        issues ??= [];
        for (const issue of result.issues)
          issues.push({ ...issue, path: [key, ...(issue.path ?? [])] });
      } else if (result.value !== undefined || own(raw, key)) {
        if (key === "__proto__")
          Object.defineProperty(output, key, {
            value: result.value,
            enumerable: true,
            writable: true,
            configurable: true,
          });
        else output[key] = result.value;
      }
    }
    if (mode !== "strip")
      for (const key of Object.keys(raw))
        if (!own(fields, key)) {
          if (mode === "strict")
            (issues ??= []).push({
              message: `Unknown key: ${key}`,
              code: "unrecognized_key",
              path: [key],
            });
          else
            Object.defineProperty(output, key, {
              value: raw[key],
              enumerable: true,
              writable: true,
              configurable: true,
            });
        }
    return issues ? { issues } : { value: output as ObjectOutput<S> };
  };
  return {
    meta: { kind: "object", shape: fields, unknownKeys: mode, description: options.description },
    "~standard": { version: 1, vendor: "loculus", validate },
    shape: fields as S,
  } as unknown as ObjectSchema<S>;
}
function appendCompiledIssues(issues: Issue[] | undefined, errors: readonly Issue[], path: PropertyKey[]): Issue[] {
  issues ??= [];
  for (const issue of errors) issues.push({ ...issue, path: [...path, ...(issue.path ?? [])] });
  return issues;
}
function compiledCheck(plan: FieldPlan | undefined, value: string): string {
  if (!plan || plan.enum) return "false";
  const options = plan.options as StringOptions & NumberOptions;
  const checks: string[] = [];
  if (["min", "max", "gt", "gte", "lt", "lte"].some((key) => {
    const bound = (options as Record<string, unknown>)[key];
    return bound !== undefined && typeof bound !== "number";
  })) return "false";
  if (plan.kind === "string") {
    if (options.trim || options.pattern) return "false";
    checks.push(`typeof ${value}==="string"`);
    if (options.min !== undefined) checks.push(`!(${value}.length<${String(options.min)})`);
    if (options.max !== undefined) checks.push(`!(${value}.length>${String(options.max)})`);
  } else if (plan.kind === "number") {
    const finite = options.integer ? `Number.isSafeInteger(${value})` : `Number.isFinite(${value})`;
    checks.push(`(typeof ${value}==="string"?(${value}.length===1?((${value}=${value}.charCodeAt(0)-48)>=0&&${value}<10):decimal.test(${value})&&(${value}=+${value},${finite})):typeof ${value}==="number"&&${finite})`);
    for (const [key, operator] of [["gt", "<="], ["gte", "<"], ["lt", ">="], ["lte", ">"]] as const)
      if (options[key] !== undefined) {
        const bound = options[key];
        if (options.integer && ((key === "gt" && bound < -Number.MAX_SAFE_INTEGER) ||
            (key === "gte" && bound <= -Number.MAX_SAFE_INTEGER) ||
            (key === "lt" && bound > Number.MAX_SAFE_INTEGER) ||
            (key === "lte" && bound >= Number.MAX_SAFE_INTEGER))) continue;
        checks.push(`!(${value}${operator}${String(bound)})`);
      }
  } else if (plan.kind === "boolean") {
    return `typeof ${value}==="boolean"||(${value}==="false"||${value}==="0"?(${value}=false,true):${value}==="true"||${value}==="1"?(${value}=true,true):false)`;
  } else return "false";
  return checks.join("&&");
}
type CompiledContext = { issues?: Issue[]; missing?: Set<string> };
function compiledValue(result: Result<any>, key: string, raw: object, context: CompiledContext): unknown {
  if (result.issues) context.issues = appendCompiledIssues(context.issues, result.issues, [key]);
  else if (result.value !== undefined || own(raw, key)) return result.value;
  else (context.missing ??= new Set()).add(key);
}

/* @__NO_SIDE_EFFECTS__ */
export function compile<S extends Shape>(s: ObjectSchema<S>): ObjectSchema<S> {
  if (s.meta.unknownKeys !== "strip" || s.meta.inner) return s;
  const keys = Object.keys(s.shape);
  const validators = keys.map((key) => s.shape[key]!["~standard"].validate);
  try {
    const plans = validators.map((validate) => (validate as PlannedValidate)[fieldPlan]);
    const compiledValidators = validators.map((validate, i) => plans[i]?.kind === "array" ? compileArray(plans[i]!, validate) : validate);
    const statements = keys.map((key, i) => {
      const property = JSON.stringify(key);
      const prefix = `const input${i}=Object.prototype.hasOwnProperty.call(raw,${property})?raw[${property}]:undefined;let v${i}=input${i};`;
      if (plans[i]?.kind === "array") return `${prefix}const result${i}=validate${i}(input${i});
if(result${i}.issues)v${i}=resolve(result${i},${property},raw,context??={issues:undefined,missing:undefined});
else{v${i}=result${i}.value;if(v${i}===undefined&&!Object.prototype.hasOwnProperty.call(raw,${property})){context??={issues:undefined,missing:undefined};(context.missing??=new Set()).add(${property});}}`;
      return `${prefix}
if(!(${compiledCheck(plans[i], `v${i}`)}))v${i}=resolve(validate${i}(input${i}),${property},raw,context??={issues:undefined,missing:undefined});`;
    });
    const properties = keys.map((key, i) => `${key === "__proto__" ? `[${JSON.stringify(key)}]` : JSON.stringify(key)}:v${i}`).join(",");
    const additions = keys.map((key, i) => {
      const assignment = key === "__proto__"
        ? `Object.defineProperty(output,${JSON.stringify(key)},{value:v${i},enumerable:true,writable:true,configurable:true});`
        : `output[${JSON.stringify(key)}]=v${i};`;
      return `if(!context.missing.has(${JSON.stringify(key)})){${assignment}}`;
    }).join("");
    const validate = new Function("fail", "decimal", "resolve", ...keys.map((_, i) => `validate${i}`),
      `return function(raw){
if(typeof raw!=="object"||raw===null||Array.isArray(raw))return fail("Expected an object","invalid_type");
let context;${statements.join("\n")}
if(context?.issues)return {issues:context.issues};
if(!context?.missing)return {value:{${properties}}};
const output={};${additions}return {value:output};}`,
    )(fail, decimal, compiledValue, ...compiledValidators);
    return { ...s, "~standard": { ...s["~standard"], validate } };
  } catch {
    return s;
  }
}
/* @__NO_SIDE_EFFECTS__ */
export function union<const S extends readonly [AnySchema, AnySchema, ...AnySchema[]]>(
  options: S,
  config: { description?: string } = {},
): Schema<Input<S[number]>, Output<S[number]>> {
  const branches = Object.freeze([...options]);
  return schema({ kind: "union", options: branches, description: config.description }, (input) => {
    const issues: Issue[] = [];
    for (const s of branches) {
      const result = s["~standard"].validate(input);
      if (!result.issues) return result;
      issues.push(...result.issues);
    }
    return { issues };
  });
}
/* @__NO_SIDE_EFFECTS__ */
export function xor<const S extends readonly [AnySchema, AnySchema, ...AnySchema[]]>(
  options: S,
  config: { description?: string } = {},
): Schema<Input<S[number]>, Output<S[number]>> {
  const branches = Object.freeze([...options]);
  return schema({ kind: "xor", options: branches, description: config.description }, (input) => {
    let success: Result<Output<S[number]>> | undefined;
    for (const s of branches) {
      const result = s["~standard"].validate(input);
      if (!result.issues) {
        if (success) return fail("Expected exactly one matching schema");
        success = result;
      }
    }
    return success ?? fail("No matching schema");
  });
}
/* @__NO_SIDE_EFFECTS__ */
export function discriminatedUnion<
  const S extends readonly [ObjectSchema, ObjectSchema, ...ObjectSchema[]],
>(
  key: string,
  options: S,
  config: { description?: string } = {},
): Schema<Input<S[number]>, Output<S[number]>> {
  const lookup = new Map<string, AnySchema>();
  for (const option of options) {
    const values = option.shape[key]?.meta.enumValues;
    if (!values?.length) throw new TypeError("Discriminator must be an enum or literal");
    for (const value of values) {
      if (lookup.has(value)) throw new TypeError(`Duplicate discriminator: ${value}`);
      lookup.set(value, option);
    }
  }
  return schema(
    {
      kind: "union",
      discriminator: key,
      options: Object.freeze([...options]),
      description: config.description,
    },
    (input) => {
      if (typeof input !== "object" || input === null || !own(input, key))
        return fail(`Missing discriminator: ${key}`);
      const selected = lookup.get((input as Record<string, string>)[key]!);
      return selected
        ? selected["~standard"].validate(input)
        : { issues: [{ message: `Invalid discriminator: ${key}`, path: [key] }] };
    },
  );
}
/* @__NO_SIDE_EFFECTS__ */
export function intersection<A extends AnySchema, B extends AnySchema>(
  a: A,
  b: B,
  config: { description?: string } = {},
): Schema<Input<A> & Input<B>, Output<A> & Output<B>> {
  return schema(
    { kind: "intersection", options: [a, b], description: config.description },
    (input) => {
      const left = a["~standard"].validate(input);
      const right = b["~standard"].validate(input);
      if (left.issues || right.issues)
        return { issues: [...(left.issues ?? []), ...(right.issues ?? [])] };
      if (
        typeof left.value !== "object" ||
        left.value === null ||
        typeof right.value !== "object" ||
        right.value === null
      )
        return fail("Intersections require object outputs");
      for (const key of Object.keys(left.value))
        if (own(right.value, key) && !Object.is(left.value[key], right.value[key]))
          return fail(`Conflicting intersection output: ${key}`);
      return { value: { ...left.value, ...right.value } };
    },
  );
}
export function fromEntries(
  entries: Iterable<readonly [string, unknown]>,
): Record<string, unknown> {
  const result: Record<string, unknown> = Object.create(null);
  for (const [key, value] of entries) {
    if (own(result, key)) {
      const current = result[key];
      if (Array.isArray(current)) current.push(value);
      else result[key] = [current, value];
    } else result[key] = value;
  }
  return result;
}
